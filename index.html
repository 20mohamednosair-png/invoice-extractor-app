import os import re import pandas as pd import pypdf # Config Paths STORE_EXCEL = "Uae Stores updated- 2026.xlsx" LIST_EXCEL = "test 4.xlsx" PDF_FOLDER = "path/to/your/pdf/folder" POSTING_DATE = "08/07/2026" # 1. Load Stores Mapping df_stores = pd.read_excel(STORE_EXCEL) store_map = {} for _, row in df_stores.iterrows(): if pd.notna(row.get('Store')) and pd.notna(row.get('Store Code')): clean_name = re.sub(r'^\d+\s+', '', str(row['Store'])).strip().upper() store_map[clean_name] = str(row['Store Code']) # 2. Load Invoice Targets df_list = pd.read_excel(LIST_EXCEL) required_invs = {} for _, row in df_list.iterrows(): inv_no = str(row.get('Bill Number') or row.get('Invoice Number')).strip() required_invs[inv_no] = str(row.get('Store', '')).strip().upper() # 3. Read PDFs Fast extracted_data = {} for filename in os.listdir(PDF_FOLDER): if filename.endswith(".pdf"): path = os.path.join(PDF_FOLDER, filename) try: reader = pypdf.PdfReader(path) text = reader.pages[0].extract_text() # Page 1 only bill_match = re.search(r'(?:Bill|Invoice)\s*No[\.\s:]*([A-Z0-9\/-]{5,20})', text, re.I) charge_match = re.search(r'(?:Current\s*Month\s*Charges|Total\s*Current\s*Charges)[\.\s:]*(?:AED)?\s*([\d,]+\.\d{2})', text, re.I) if bill_match and charge_match: bill_no = bill_match.group(1).strip() amount = float(charge_match.group(1).replace(',', '')) extracted_data[bill_no] = amount except Exception: continue # 4. Generate SAP Output Data output = [] for bill_no, store_name in required_invs.items(): if bill_no in extracted_data: gross = extracted_data[bill_no] net = round(gross / 1.05, 2) store_code = store_map.get(store_name, "").zfill(4) cost_center = f"AE{store_code}610" if store_code else "" output.append({ "Company Code": "AE03", "Transaction": 1, "Invoicing Party": "2000004275", "Reference": bill_no, "Posting Date": POSTING_DATE, "Document Type": "KR", "Gross Invoice Amount": f"{gross:.2f}", "Amount in Document Currency": f"{net:.2f}", "Tax Code": "IT", "Cost Center": cost_center, "Profit Center": "100005" }) df_output = pd.DataFrame(output) df_output.to_excel("SAP_Upload.xlsx", index=False) print(f"Done! Created SAP_Upload.xlsx with {len(df_output)} records.")
